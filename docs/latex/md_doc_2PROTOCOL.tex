{\bfseries{}}\hypertarget{md_doc_2PROTOCOL_autotoc_md60}{}\doxysection{\texorpdfstring{Communication Flow}{Communication Flow}}\label{md_doc_2PROTOCOL_autotoc_md60}
{\bfseries{}}

{\bfseries{
\begin{DoxyEnumerate}
\item {\bfseries{Connection Phase}}\+: ~\newline

\begin{DoxyItemize}
\item Client sends {\ttfamily NEW\+\_\+\+CLIENT} to the server. ~\newline

\item \doxylink{classServer}{Server} responds with {\ttfamily SEND\+\_\+\+UUID} to assign a unique ID.
\end{DoxyItemize}
\item {\bfseries{Game Preparation Phase}}\+: ~\newline

\begin{DoxyItemize}
\item Client sends {\ttfamily CLIENT\+\_\+\+READY} with player details when ready. ~\newline

\item \doxylink{classServer}{Server} sends {\ttfamily READY\+\_\+\+SIGNAL} to all ready clients when the game is ready to start.
\end{DoxyItemize}
\item {\bfseries{Gameplay Phase}}\+: ~\newline

\begin{DoxyItemize}
\item \doxylink{classServer}{Server} sends {\ttfamily CREATE\+\_\+\+ENTITY} messages to initialize game entities on the client. ~\newline

\item Client sends {\ttfamily CLIENT\+\_\+\+INPUT} messages to notify the server of player actions. ~\newline

\item \doxylink{classServer}{Server} periodically sends {\ttfamily UPDATE\+\_\+\+ENTITY} messages to synchronize entity states.
\end{DoxyItemize}
\item {\bfseries{Disconnection/\+End Phase}}\+: ~\newline

\begin{DoxyItemize}
\item Client can send {\ttfamily CLIENT\+\_\+\+DISCONNECT} to leave the game. ~\newline

\item \doxylink{classServer}{Server} can send {\ttfamily GAME\+\_\+\+OVER} to notify clients that the game has ended.
\end{DoxyItemize}
\end{DoxyEnumerate}}}
